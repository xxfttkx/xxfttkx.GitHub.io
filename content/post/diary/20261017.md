int exGaugeRate = data.GetBaseExGaugeRate();

// --- 累计 6 类技能效果的加成值 ---
int bonus = 0;
foreach (var s in data.GetSkillEffect(SkillType.ExRateUp))           bonus += s.SkillValue1;    // +0x34
foreach (var s in data.GetSkillEffect(SkillType.ExRateDown))         bonus -= s.SkillValue1;    // -0x34

foreach (var s in data.GetSkillEffect(SkillType.ExRateUpAttrOut))    bonus += s.SkillEffectValue; // +0x70
foreach (var s in data.GetSkillEffect(SkillType.ExRateUpSkillCount)) bonus += s.SkillEffectValue; // +0x70
foreach (var s in data.GetSkillEffect(SkillType.ExRateUpSpecific))   bonus += s.SkillValue1;    // +0x34

// --- 上限裁剪 ---
bonus = Math.Min(bonus, 300);

// --- 与基础值合成 ---
int total = Math.Max(0, exGaugeRate + bonus);

// --- 最终公式 ---
int result = (int)(baseValue * (1.0f + (total - 50.0f) / 150.0f));
